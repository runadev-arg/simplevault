**`credentials-client.ts`**:
```ts
import { apiFetch } from "./api-fetch";
import { bytesToB64Url, b64UrlToBytes } from "./base64url";

export interface CredentialBlob { ciphertext: Uint8Array; nonce: Uint8Array; aadParamsJson: string; }
export interface CredentialResponse extends CredentialBlob { id: string; vaultId: string; version: number; updatedAt: string; }
export interface CredentialSummary { id: string; version: number; updatedAt: string; }
export interface VaultPersonalResponse { vaultId: string; credentialIds: CredentialSummary[]; }

export async function listVaultPersonal(): Promise<VaultPersonalResponse> {
  const r = await apiFetch("/vault/personal", { method: "GET" });
  return r as VaultPersonalResponse;
}

export async function createCredential(input: { vaultId: string; ciphertext: Uint8Array; nonce: Uint8Array; aadParamsJson: string }): Promise<{ id: string; vaultId: string; version: number; updatedAt: string }> {
  return apiFetch("/credentials", {
    method: "POST",
    body: JSON.stringify({
      vaultId: input.vaultId,
      ciphertext: bytesToB64Url(input.ciphertext),
      nonce: bytesToB64Url(input.nonce),
      aadParamsJson: input.aadParamsJson,
      version: 1,
    }),
  });
}

export async function getCredential(id: string): Promise<CredentialResponse> {
  const raw = await apiFetch(`/credentials/${id}`, { method: "GET" }) as { id: string; vaultId: string; version: number; ciphertext: string; nonce: string; aadParamsJson: string; updatedAt: string };
  return {
    id: raw.id, vaultId: raw.vaultId, version: raw.version,
    ciphertext: b64UrlToBytes(raw.ciphertext),
    nonce: b64UrlToBytes(raw.nonce),
    aadParamsJson: raw.aadParamsJson, updatedAt: raw.updatedAt,
  };
}

export async function updateCredential(id: string, input: { ciphertext: Uint8Array; nonce: Uint8Array; aadParamsJson: string; version: number }): Promise<{ id: string; vaultId: string; version: number; updatedAt: string }> {
  return apiFetch(`/credentials/${id}`, {
    method: "PATCH",
    body: JSON.stringify({
      ciphertext: bytesToB64Url(input.ciphertext),
      nonce: bytesToB64Url(input.nonce),
      aadParamsJson: input.aadParamsJson,
      version: input.version,
    }),
  });
}

export async function deleteCredential(id: string): Promise<void> {
  await apiFetch(`/credentials/${id}`, { method: "DELETE" });
}
```

**`errors.ts`** — add subclasses:
```ts
export class VersionConflictError extends ApiError {
  readonly credentialId: string;
  constructor(credentialId: string, message = "Credential version conflict") {
    super(message, "CREDENTIAL_VERSION_CONFLICT", 409);
    this.credentialId = credentialId;
  }
}
export class CredentialNotFoundError extends ApiError { constructor() { super("Credential not found", "CREDENTIAL_NOT_FOUND", 404); } }
export class CredentialBodyTooLargeError extends ApiError { constructor() { super("Credential body exceeds 64 KiB", "CREDENTIAL_BODY_TOO_LARGE", 413); } }
```

**`api-fetch.ts`** error-mapping extension — in the existing `if (!res.ok)` branch, after parsing the body's `error.code`:
```ts
switch (body?.error?.code) {
  case "CREDENTIAL_VERSION_CONFLICT": throw new VersionConflictError(/* extract from URL or body */);
  case "CREDENTIAL_NOT_FOUND": throw new CredentialNotFoundError();
  case "CREDENTIAL_BODY_TOO_LARGE": throw new CredentialBodyTooLargeError();
  default: throw new ApiError(body?.error?.message ?? "Request failed", body?.error?.code ?? "UNKNOWN", res.status);
}
```

For `VersionConflictError.credentialId`, the apiFetch caller passes the URL to apiFetch; the switch can extract `:id` from the URL OR `updateCredential` can wrap the apiFetch call with a try/catch that re-throws with the id. Choose the wrapper-rewrap approach (cleaner separation):
```ts
export async function updateCredential(id: string, input: ...) {
  try { return await apiFetch(...); }
  catch (e) {
    if (e instanceof ApiError && e.code === "CREDENTIAL_VERSION_CONFLICT") throw new VersionConflictError(id, e.message);
    throw e;
  }
}
```
