**`use-auto-lock.ts`**:
```ts
"use client";
import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { keyStore } from "@/lib/crypto/key-store";
import { accessTokenStore } from "@/lib/auth/access-token-store"; // or whatever the Phase-02 in-memory JWT store is

const IDLE_MS = (() => {
  const raw = process.env.NEXT_PUBLIC_AUTO_LOCK_IDLE_MS;
  const n = raw ? Number.parseInt(raw, 10) : NaN;
  return Number.isFinite(n) && n > 0 ? n : 900_000;
})();

export function useAutoLock(): void {
  const router = useRouter();
  useEffect(() => {
    let timer: ReturnType<typeof setTimeout> | null = null;
    const onIdle = () => {
      keyStore.wipe();
      accessTokenStore.clear();
      router.replace("/login?reason=auto_lock");
    };
    const reset = () => {
      if (timer) clearTimeout(timer);
      timer = setTimeout(onIdle, IDLE_MS);
    };
    const events: Array<keyof DocumentEventMap> = ["pointerdown", "keydown", "visibilitychange"];
    for (const e of events) document.addEventListener(e, reset, { passive: true });
    reset(); // start the timer on mount
    return () => {
      if (timer) clearTimeout(timer);
      for (const e of events) document.removeEventListener(e, reset);
    };
  }, [router]);
}
```

Note: per operator decision 4, `visibilitychange` RESETS the timer (treat as activity), it does NOT start a tighter deadline.
