**`create`** — accepts client-supplied `credentialId` (Plan 04-10 cross-dep) via COALESCE:
```ts
// Verify vault belongs to user; insert in same statement (single SQL via INSERT ... SELECT)
const inserted = await this.db.execute<CredentialRow>(sql`
  INSERT INTO credentials (id, vault_id, version, ciphertext, nonce, aad_params_json)
  SELECT COALESCE(${dto.credentialId ?? null}::uuid, gen_random_uuid()),
         v.id, 1, ${dto.ciphertext}, ${dto.nonce}, ${dto.aadParamsJson}
  FROM vaults v
  WHERE v.id = ${dto.vaultId} AND v.owner_user_id = ${userId}
  RETURNING id, vault_id, version, updated_at
`);
if (inserted.rows.length === 0) throw notFound(); // vault doesn't belong to user
return shape(inserted.rows[0]);
```

**`getById`** — single SELECT with ownership join:
```ts
const r = await this.db.execute<CredentialRow>(sql`
  SELECT c.id, c.vault_id, c.version, c.ciphertext, c.nonce, c.aad_params_json, c.updated_at
  FROM credentials c
  JOIN vaults v ON v.id = c.vault_id
  WHERE c.id = ${id} AND v.owner_user_id = ${userId}
`);
if (r.rows.length === 0) throw notFound();
return shape(r.rows[0]);
```

**`update`** — atomic CAS:
```ts
const r = await this.db.execute<CredentialRow>(sql`
  UPDATE credentials c
  SET ciphertext = ${dto.ciphertext},
      nonce = ${dto.nonce},
      aad_params_json = ${dto.aadParamsJson},
      version = c.version + 1,
      updated_at = now()
  FROM vaults v
  WHERE c.id = ${id}
    AND v.id = c.vault_id
    AND v.owner_user_id = ${userId}
    AND c.version = ${dto.version}
  RETURNING c.id, c.vault_id, c.version, c.updated_at
`);
if (r.rows.length === 0) {
  // Disambiguate: was it ownership (404) or version (409)?
  const exists = await this.db.execute<{c: number}>(sql`
    SELECT 1 FROM credentials c JOIN vaults v ON v.id = c.vault_id
    WHERE c.id = ${id} AND v.owner_user_id = ${userId}
  `);
  throw exists.rows.length === 0 ? notFound() : versionConflict();
}
return shape(r.rows[0]);
```

Subtle but load-bearing: the post-failure SELECT runs ONLY when the CAS fails. It cannot leak existence cross-user because it joins `owner_user_id`. If the row exists for THIS user but the version mismatched, return 409. If the row doesn't exist or belongs to another user, return 404. This is the uniform-404 anti-enum guarantee.

**`delete`**:
```ts
const r = await this.db.execute(sql`
  DELETE FROM credentials c USING vaults v
  WHERE c.id = ${id} AND v.id = c.vault_id AND v.owner_user_id = ${userId}
  RETURNING c.id
`);
if (r.rowCount === 0) throw notFound();
```

**`credentials.dto.ts`** — `.strict()` Zod (Truth 18):
```ts
export const CredentialCreateSchema = z.object({
  vaultId: z.string().uuid(),
  // Plan 04-10 cross-dep: client supplies the credentialId so AAD is self-consistent on POST
  // (avoids a two-roundtrip "encrypt with placeholder, server returns id, re-encrypt" dance).
  // Server uses COALESCE($credentialId, gen_random_uuid()).
  credentialId: z.string().uuid().optional(),
  ciphertext: z.instanceof(Uint8Array).refine(b => b.length > 0 && b.length <= 65536),
  nonce: z.instanceof(Uint8Array).refine(b => b.length === 24, "nonce must be 24 bytes"),
  aadParamsJson: z.string().max(2048),
  version: z.literal(1),
}).strict();

export const CredentialUpdateSchema = z.object({
  ciphertext: z.instanceof(Uint8Array).refine(b => b.length > 0 && b.length <= 65536),
  nonce: z.instanceof(Uint8Array).refine(b => b.length === 24),
  aadParamsJson: z.string().max(2048),
  version: z.number().int().nonnegative(),
}).strict();

export const CredentialResponseSchema = z.object({
  id: z.string().uuid(),
  vaultId: z.string().uuid(),
  version: z.number().int(),
  ciphertext: z.instanceof(Uint8Array).optional(), // optional for the create-response shape (id+version only)
  nonce: z.instanceof(Uint8Array).optional(),
  aadParamsJson: z.string().optional(),
  updatedAt: z.string(), // ISO
});
```
Bytea handling: bodies arrive as base64-encoded JSON or raw `application/octet-stream`. Decision: accept JSON with base64url-encoded `ciphertext` + `nonce`; the DTO `transform()` decodes to `Uint8Array`. This matches the existing Phase-03 `wrappedSecret` posture.

**`credentials.controller.ts`**:
```ts
@Controller()
@UseGuards(JwtAuthGuard) // already APP_GUARD; explicit for clarity
export class CredentialsController {
  constructor(private readonly svc: CredentialsService) {}

  @Post("credentials")
  @Throttle({ "credentials-write-user": { limit: 60, ttl: 60_000 } }) // declared in Plan 04-03
  async create(@CurrentUser() user: AuthUser, @Body(zodPipe(CredentialCreateSchema)) dto: CreateCredentialDto) {
    const r = await this.svc.create(user.id, dto);
    AuditEventService.emit(this.logger, { action: "credential.create", actorUserId: user.id, targetId: r.id, outcome: "ok", data: { credentialId: r.id, vaultId: r.vaultId, version: r.version } });
    return r;
  }

  @Get("credentials/:id")
  @Throttle({ "credentials-read-user": { limit: 300, ttl: 60_000 } })
  async getById(@CurrentUser() user: AuthUser, @Param("id", ParseUUIDPipe) id: string) {
    const r = await this.svc.getById(user.id, id);
    AuditEventService.emit(this.logger, { action: "credential.view", actorUserId: user.id, targetId: r.id, outcome: "ok", data: { credentialId: r.id, vaultId: r.vaultId, version: r.version } });
    return r;
  }

  @Patch("credentials/:id")
  @Throttle({ "credentials-write-user": { limit: 60, ttl: 60_000 } })
  async update(@CurrentUser() user: AuthUser, @Param("id", ParseUUIDPipe) id: string, @Body(zodPipe(CredentialUpdateSchema)) dto: UpdateCredentialDto) {
    const r = await this.svc.update(user.id, id, dto);
    AuditEventService.emit(this.logger, { action: "credential.update", actorUserId: user.id, targetId: r.id, outcome: "ok", data: { credentialId: r.id, vaultId: r.vaultId, version: r.version } });
    return r;
  }

  @Delete("credentials/:id")
  @HttpCode(204)
  @Throttle({ "credentials-write-user": { limit: 60, ttl: 60_000 } })
  async delete(@CurrentUser() user: AuthUser, @Param("id", ParseUUIDPipe) id: string) {
    await this.svc.delete(user.id, id);
    AuditEventService.emit(this.logger, { action: "credential.delete", actorUserId: user.id, targetId: id, outcome: "ok", data: { credentialId: id } });
  }
}
```
