**Implementation**:
```ts
import { EFF_LARGE } from "./eff-large";

const LOWER = "abcdefghijklmnopqrstuvwxyz";
const UPPER = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
const DIGITS = "0123456789";
const SYMBOLS = "!@#$%^&*()-_=+[]{};:,.<>/?"; // 25 chars; document the choice
// Total when all on: 26+26+10+25 = 87 chars. (Note: not 94 — the spec said ~94. Operator can extend SYMBOLS later. Document the conservative choice in SUMMARY: log2(87) ≈ 6.44, 20 chars ≈ 128.8 bits — still above the 128-bit anchor.)

function randInt(maxExclusive: number): number {
  // Rejection sampling for unbiased uniform in [0, maxExclusive)
  if (maxExclusive <= 0 || maxExclusive > 256) throw new RangeError("randInt range");
  const buf = new Uint8Array(1);
  const limit = 256 - (256 % maxExclusive); // largest multiple of maxExclusive ≤ 256
  // eslint-disable-next-line no-constant-condition
  while (true) {
    crypto.getRandomValues(buf);
    if (buf[0] < limit) return buf[0] % maxExclusive;
  }
}

function randIndex16(maxExclusive: number): number {
  // For wordlist (7776) — need a 16-bit primitive
  if (maxExclusive <= 0 || maxExclusive > 65536) throw new RangeError();
  const buf = new Uint16Array(1);
  const limit = 65536 - (65536 % maxExclusive);
  // eslint-disable-next-line no-constant-condition
  while (true) {
    crypto.getRandomValues(buf);
    if (buf[0] < limit) return buf[0] % maxExclusive;
  }
}

export function generatePassword(opts: PasswordOptions): string {
  if (opts.length < 8 || opts.length > 128) throw new RangeError("length must be 8..128");
  const enabled = [
    opts.lower ? LOWER : "",
    opts.upper ? UPPER : "",
    opts.digits ? DIGITS : "",
    opts.symbols ? SYMBOLS : "",
  ].filter(s => s.length > 0);
  if (enabled.length === 0) throw new RangeError("at least one character class required");
  const all = enabled.join("");
  // First, pick one char from each enabled class (guarantees class invariant).
  const result: string[] = enabled.map(cls => cls[randInt(cls.length)]);
  // Fill the rest from the union charset.
  while (result.length < opts.length) result.push(all[randInt(all.length)]);
  // Fisher–Yates shuffle (so the guaranteed chars aren't always at the front).
  for (let i = result.length - 1; i > 0; i--) {
    const j = randInt(i + 1);
    [result[i], result[j]] = [result[j], result[i]];
  }
  return result.join("");
}

export function generatePassphrase(opts: PassphraseOptions): string {
  if (opts.wordCount < 3 || opts.wordCount > 10) throw new RangeError("wordCount must be 3..10");
  const words: string[] = [];
  for (let i = 0; i < opts.wordCount; i++) words.push(EFF_LARGE[randIndex16(EFF_LARGE.length)]);
  return words.join(opts.separator);
}
```
