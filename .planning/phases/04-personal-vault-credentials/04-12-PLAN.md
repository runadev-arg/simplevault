```ts
describe("credentials CRUD", () => {
  it("create → list → view → edit → 409 race → delete (single user)", () => {
    cy.signupAndLogin("alice@example.com", "MasterPw1!");
    cy.visit("/vault");
    cy.contains(/no credentials yet|empty/i);
    cy.contains(/new credential|create/i).click();
    cy.url().should("include", "/credential/new");
    cy.get("input[name=name]").type("GitHub");
    cy.get("input[name=urls]").type("https://github.com");
    cy.get("input[name=username]").type("alice");
    cy.get("input[name=password]").type("hunter2hunter2hunter2");
    cy.contains("Save").click();
    cy.url().should("include", "/vault");
    cy.contains("GitHub").click();
    cy.url().should("match", /\/credential\/[0-9a-f-]{36}/);
    cy.get("input[name=password]").clear().type("newP@ssw0rdLong!Long!");
    cy.contains("Save").click();
    cy.url().should("include", "/vault");
    // 409 race: PATCH with stale version via apiFetch directly
    cy.window().then(async (win) => {
      // Find the credential id from the URL or list, then issue a PATCH with version=1 (stale; current is 2)
      const list = await (win as any).__test_listVaultPersonal();
      const credId = list.credentialIds[0].id;
      const r = await fetch(`/credentials/${credId}`, { method: "PATCH", headers: { "content-type": "application/json", authorization: `Bearer ${win.__test_accessToken}` }, body: JSON.stringify({ ciphertext: "AAA", nonce: "A".repeat(32), aadParamsJson: "{}", version: 1 }) });
      expect(r.status).to.eq(409);
      const body = await r.json();
      expect(body.error.code).to.eq("CREDENTIAL_VERSION_CONFLICT");
    });
    // Delete
    cy.contains("GitHub").click();
    cy.contains("Delete").click();
    cy.on("window:confirm", () => true);
    cy.url().should("include", "/vault");
    cy.contains(/no credentials yet|empty/i);
  });

  it("cross-user IDOR returns 404 not 403", () => {
    cy.signupAndLogin("alice@example.com", "MasterPw1!");
    cy.makeCredential("GitHub", "hunter2hunter2hunter2");
    cy.window().then((w) => { (w as any).__alice = { token: w.__test_accessToken, credId: w.__test_lastCredId }; });
    cy.signupAndLogin("bob@example.com", "MasterPw2!");
    cy.window().then(async (win) => {
      const r = await fetch(`/credentials/${(cy as any).state("alice").credId}`, { headers: { authorization: `Bearer ${win.__test_accessToken}` } });
      expect(r.status).to.eq(404);
      const body = await r.json();
      expect(body.error.code).to.eq("CREDENTIAL_NOT_FOUND");
    });
  });

  it("rejects > 64 KiB body with 413 CREDENTIAL_BODY_TOO_LARGE", () => {
    cy.signupAndLogin("carol@example.com", "MasterPw3!");
    cy.window().then(async (win) => {
      const huge = "A".repeat(70 * 1024); // 70 KiB
      const r = await fetch("/credentials", { method: "POST", headers: { "content-type": "application/json", authorization: `Bearer ${win.__test_accessToken}` }, body: JSON.stringify({ vaultId: "00000000-0000-0000-0000-000000000000", ciphertext: huge, nonce: "A".repeat(32), aadParamsJson: "{}", version: 1 }) });
      expect(r.status).to.eq(413);
    });
  });
});
```

**`verify-grep.sh`**:
```sh
#!/usr/bin/env bash
set -euo pipefail
fail=0

# (a) FINDING-0041 carry: server never sees master_DEK except in the redact wildcard
hits=$(grep -rEn 'master_DEK|master_dek|master_kek' apps/api/src | grep -vE 'redact|\bcensor|\.test\.' || true)
if [[ -n "$hits" ]]; then echo "GATE FAIL (a): server-side master_DEK references"; echo "$hits"; fail=1; fi

# (b) Credentials/vault modules don't call @simplevault/crypto encrypt/decrypt
hits=$(grep -rEn '@simplevault/crypto' apps/api/src/credentials apps/api/src/vault || true)
if [[ -n "$hits" ]]; then echo "GATE FAIL (b): credentials/vault module imports @simplevault/crypto"; echo "$hits"; fail=1; fi

# (c) Server has zero awareness of password sha256 / reuse
hits=$(grep -rEn 'sha256|reuse' apps/api/src/credentials apps/api/src/vault || true)
if [[ -n "$hits" ]]; then echo "GATE FAIL (c): server-side reuse/sha256 references"; echo "$hits"; fail=1; fi

# (d) 2FA guard NOT wired to personal vault routes
hits=$(grep -rEn 'Require2FAGuard|_2fa-guard-probe' apps/api/src/credentials apps/api/src/vault || true)
if [[ -n "$hits" ]]; then echo "GATE FAIL (d): 2FA guard / probe leaked into personal vault"; echo "$hits"; fail=1; fi

# (e) AAD label parity: only aad-labels.ts contains the literal
hits=$(grep -rEn '"sv:[^"]+v1\|"' apps/web/src/lib/crypto/ | grep -v 'aad-labels.ts' || true)
if [[ -n "$hits" ]]; then echo "GATE FAIL (e): AAD label literal duplicated outside aad-labels.ts"; echo "$hits"; fail=1; fi

# (f) sensitive field names not in stray code paths (informational; allow whitelist comment)
# (this is the looser gate from Plan 04-11; redact list is the safety net)
hits=$(grep -rEn '"password"|"notes"|"custom_fields"|"history"' apps/api/src --include='*.ts' | grep -vE 'redact|Schema|interface |type |//.*allow-grep' || true)
if [[ -n "$hits" ]]; then echo "GATE WARN (f): sensitive field names in api code (review):"; echo "$hits"; fi

exit $fail
```

```md
## Phase 04 — Personal vault: credentials

### New environment variables
| Variable | Default | Purpose |
| --- | --- | --- |
| `NEXT_PUBLIC_AUTO_LOCK_IDLE_MS` | `900000` (15 min) | Web idle timeout before client-side auto-lock |
| `CREDENTIAL_BODY_MAX_BYTES` | `65536` (constant; future env) | Per-route body cap on `/credentials/*` |

### Operational notes
- **Auto-lock is CLIENT-ONLY.** When the user is auto-locked, master_DEK is wiped from browser RAM and the user is redirected to `/login?reason=auto_lock`. The server-side session remains valid until refresh-token rotation expires. To revoke ALL remote sessions, the user must call `POST /sessions/revoke-all` (Phase-03 endpoint) — this is a separate, deliberate action.
- **CSP — libsodium WASM:** continues to load via `'self'` (REQ-WEBSEC-001). No new CSP directive needed.
- **Phase-03 `_2fa-guard-probe` route DELETED.** Plan 04-01 retired the stub. The `Require2FAGuard` source is preserved for Phase 07 (shared vaults).
- **Body-size cap on `/credentials/*` routes:** 64 KiB (FINDING-0015 partial closure). Other routes still use the Express default; FINDING-0015 fully closes in Phase 13.
- **Reuse detection is browser-only** and per-vault. Shared-vault reuse detection scope is a Phase-07 decision.

### Backfill on upgrade
Migration `0003_phase04_vaults_credentials.sql` includes an idempotent `INSERT INTO vaults ... WHERE NOT EXISTS` that creates a personal vault row for every existing user. Re-running the migration is a no-op.
```
