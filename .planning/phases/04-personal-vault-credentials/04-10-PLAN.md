```tsx
"use client";
import { useRouter } from "next/navigation";
import { listVaultPersonal, createCredential } from "@/lib/api/credentials-client";
import { encryptCredential } from "@/lib/crypto/credential-cipher";
import { buildVaultCredentialAad } from "@simplevault/crypto/browser";
import { AAD_LABEL_VAULT_CREDENTIAL } from "@/lib/crypto/aad-labels";
import { keyStore } from "@/lib/crypto/key-store";

export default function NewCredentialPage() {
  const router = useRouter();
  const onSave = async (d: DecryptedCredential) => {
    const { vaultId } = await listVaultPersonal();
    const credentialId = crypto.randomUUID();
    const email = keyStore.get("email");
    const masterDek = keyStore.get("master_DEK");
    const aad = buildVaultCredentialAad({ vaultId, credentialId, version: 1, email }, AAD_LABEL_VAULT_CREDENTIAL);
    const aadParamsJson = canonicalCredentialAadJson({ vaultId, credentialId, version: 1 });
    const { ciphertext, nonce } = encryptCredential(JSON.stringify(d), masterDek, aad);
    await createCredential({ vaultId, credentialId, ciphertext, nonce, aadParamsJson });
    router.replace("/vault");
  };
  return <CredentialEditor mode="new" onSave={onSave} />;
}
```

```ts
/** Best-effort clipboard clear. Browsers cannot truly purge the OS clipboard once the
 *  user pastes elsewhere; this overwrite-after-30s is a defence-in-depth stub. */
export function copyPasswordWithClear(pw: string, ms = 30_000): void {
  navigator.clipboard.writeText(pw);
  setTimeout(() => { navigator.clipboard.writeText("").catch(() => {}); }, ms);
}
```

**`/credential/[id]/page.tsx`**:
```tsx
"use client";
import { useEffect, useState } from "react";
import { useParams, useRouter } from "next/navigation";
import { getCredential, updateCredential, deleteCredential, listVaultPersonal } from "@/lib/api/credentials-client";
import { decryptCredential, encryptCredential } from "@/lib/crypto/credential-cipher";
import { buildVaultCredentialAad, canonicalCredentialAadJson } from "@simplevault/crypto/browser";
import { VersionConflictError } from "@/lib/api/errors";

export default function EditCredentialPage() {
  const { id } = useParams<{ id: string }>();
  const router = useRouter();
  const [baseline, setBaseline] = useState<{ vaultId: string; version: number; decrypted: DecryptedCredential } | null>(null);

  useEffect(() => { (async () => {
    const { vaultId } = await listVaultPersonal();
    const blob = await getCredential(id);
    const aad = buildVaultCredentialAad({ vaultId, credentialId: id, version: blob.version, email: keyStore.get("email") }, AAD_LABEL_VAULT_CREDENTIAL);
    const json = decryptCredential({ ciphertext: blob.ciphertext, nonce: blob.nonce }, keyStore.get("master_DEK"), aad);
    const decrypted = DecryptedCredentialSchema.parse(JSON.parse(json));
    setBaseline({ vaultId, version: blob.version, decrypted });
  })(); }, [id]);

  const onSave = async (next: DecryptedCredential) => {
    if (!baseline) return;
    // History rotation: if password changed, append previous to history (cap 10)
    if (next.password !== baseline.decrypted.password) {
      next.history = [{ password: baseline.decrypted.password, changedAt: new Date().toISOString() }, ...(baseline.decrypted.history ?? [])].slice(0, 10);
    } else {
      next.history = baseline.decrypted.history ?? [];
    }
    const newVersion = baseline.version + 1;
    const aad = buildVaultCredentialAad({ vaultId: baseline.vaultId, credentialId: id, version: newVersion, email: keyStore.get("email") }, AAD_LABEL_VAULT_CREDENTIAL);
    const aadParamsJson = canonicalCredentialAadJson({ vaultId: baseline.vaultId, credentialId: id, version: newVersion });
    const { ciphertext, nonce } = encryptCredential(JSON.stringify(next), keyStore.get("master_DEK"), aad);
    try {
      await updateCredential(id, { ciphertext, nonce, aadParamsJson, version: baseline.version });
      router.replace("/vault");
    } catch (e) {
      if (e instanceof VersionConflictError) {
        // Re-fetch and prompt: "This credential was updated elsewhere. Reload your edits onto the latest?"
        const fresh = await getCredential(id);
        const freshAad = buildVaultCredentialAad({ vaultId: baseline.vaultId, credentialId: id, version: fresh.version, email: keyStore.get("email") }, AAD_LABEL_VAULT_CREDENTIAL);
        const freshJson = decryptCredential({ ciphertext: fresh.ciphertext, nonce: fresh.nonce }, keyStore.get("master_DEK"), freshAad);
        const freshDecrypted = DecryptedCredentialSchema.parse(JSON.parse(freshJson));
        if (confirm("This credential was updated elsewhere. Apply your edits on top of the latest version?")) {
          setBaseline({ vaultId: baseline.vaultId, version: fresh.version, decrypted: freshDecrypted });
          return onSave(next); // recursive — uses the fresh baseline + bumps to fresh.version + 1
        }
        return;
      }
      throw e;
    }
  };

  const onDelete = async () => {
    if (!confirm("Delete this credential? This cannot be undone.")) return;
    await deleteCredential(id);
    router.replace("/vault");
  };

  if (!baseline) return <Skeleton />;
  return <CredentialEditor mode="edit" initial={baseline.decrypted} onSave={onSave} onDelete={onDelete} />;
}
```

History cap = 10: enforced by `.slice(0, 10)`. `DecryptedCredentialSchema.history.max(10)` validates on parse — if a malformed blob has > 10 entries, parse fails (server-side) — UI shows an error "Credential history corrupted; contact admin."
