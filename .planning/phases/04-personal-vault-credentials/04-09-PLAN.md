**`reuse-set.ts`**:
```ts
import sodium from "libsodium-wrappers-sumo";
await sodium.ready;
export function passwordHash(pw: string): string {
  // sha256 hex of the password — Set membership key
  return sodium.to_hex(sodium.crypto_hash_sha256(sodium.from_string(pw)));
}
export function buildReuseSet(decryptedCreds: Array<{ id: string; password: string }>): Map<string, string[]> {
  // map: hash → [credentialIds] (so the badge can show "reused with N others")
  const m = new Map<string, string[]>();
  for (const c of decryptedCreds) {
    const h = passwordHash(c.password);
    const arr = m.get(h) ?? [];
    arr.push(c.id);
    m.set(h, arr);
  }
  return m;
}
export function reuseCountForCred(map: Map<string, string[]>, credId: string, pwHash: string): number {
  const ids = map.get(pwHash) ?? [];
  return Math.max(0, ids.length - 1); // exclude self
}
```

**`credential-card.tsx`** — uses an IntersectionObserver to defer decrypt:
```tsx
export function CredentialCard({ summary, vaultId, onDecrypted, reuseMap }: Props) {
  const ref = useRef<HTMLDivElement>(null);
  const [decrypted, setDecrypted] = useState<DecryptedSummary | null>(null);
  useEffect(() => {
    const obs = new IntersectionObserver(async ([entry]) => {
      if (!entry.isIntersecting || decrypted) return;
      const blob = await getCredential(summary.id);
      const aad = buildVaultCredentialAad({ vaultId, credentialId: summary.id, version: blob.version, email: keyStore.get("email") }, AAD_LABEL_VAULT_CREDENTIAL);
      const json = decryptCredential({ ciphertext: blob.ciphertext, nonce: blob.nonce }, keyStore.get("master_DEK"), aad);
      const parsed = JSON.parse(json) as DecryptedCredential;
      const summary = { id: summary.id, name: parsed.name, urls: parsed.urls, username: parsed.username, password: parsed.password, is_favorite: !!parsed.is_favorite, updated_at: blob.updatedAt };
      setDecrypted(summary);
      onDecrypted(summary);
    }, { rootMargin: "50px" });
    if (ref.current) obs.observe(ref.current);
    return () => obs.disconnect();
  }, [summary.id, vaultId]);
  const hash = decrypted ? passwordHash(decrypted.password) : null;
  const reuseCount = hash ? reuseCountForCred(reuseMap, summary.id, hash) : 0;
  return (
    <div ref={ref} className="rounded-lg border p-4 hover:shadow-md transition">
      {decrypted ? (
        <>
          <h3 className="font-semibold">{decrypted.name}</h3>
          <p className="text-sm text-muted-foreground truncate">{decrypted.urls?.[0]}</p>
          <p className="text-xs text-muted-foreground">{decrypted.username}</p>
          <ReuseBadge count={reuseCount} />
        </>
      ) : (
        <div className="animate-pulse"><div className="h-4 bg-muted w-3/4 rounded" /><div className="h-3 bg-muted w-1/2 rounded mt-2" /></div>
      )}
    </div>
  );
}
```

The card renders `decrypted.password` ONLY in-memory (used for the hash, NOT in the DOM). Reveal happens only on the credential detail page (Plan 04-10).

```tsx
useEffect(() => {
  if (!list || list.credentialIds.length === 0) return;
  const ric = (cb: () => void) => (window as any).requestIdleCallback ? (window as any).requestIdleCallback(cb) : setTimeout(cb, 200);
  ric(async () => {
    const all = await Promise.all(list.credentialIds.map(c => getCredential(c.id)));
    const decryptedAll = all.map(blob => {
      const aad = buildVaultCredentialAad({ vaultId: list.vaultId, credentialId: blob.id, version: blob.version, email: keyStore.get("email") }, AAD_LABEL_VAULT_CREDENTIAL);
      const json = decryptCredential({ ciphertext: blob.ciphertext, nonce: blob.nonce }, keyStore.get("master_DEK"), aad);
      const parsed = JSON.parse(json) as DecryptedCredential;
      return { id: blob.id, password: parsed.password };
    });
    setReuseMap(buildReuseSet(decryptedAll));
  });
}, [list]);
```

Add a small `setReuseMap` state hook, plumb the `reuseMap` prop through to each card. Cards already-decrypted in T2 immediately reflect the new reuseMap on next render.

Memory hygiene: the bulk decrypt holds the password set in a Map<sha256, ids[]> ONLY (no plaintext passwords in long-term React state). Local `decryptedAll` array goes out of scope after `buildReuseSet` consumes it.
