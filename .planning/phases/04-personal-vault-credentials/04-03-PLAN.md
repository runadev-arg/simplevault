**`vault.service.ts`**:
```ts
@Injectable()
export class VaultService {
  constructor(@Inject(DB) private readonly db: DrizzleDb) {}
  async getPersonal(userId: string): Promise<{ vaultId: string; credentialIds: Array<{ id: string; version: number; updatedAt: string }> }> {
    return this.db.transaction(async (tx) => {
      const v = await tx.execute<{ id: string }>(sql`
        SELECT id FROM vaults WHERE owner_user_id = ${userId} AND kind = 'personal'
      `);
      if (v.rows.length === 0) {
        // Should never happen post-Plan-01 backfill; defensive auto-heal:
        const inserted = await tx.execute<{ id: string }>(sql`
          INSERT INTO vaults (owner_user_id, kind) VALUES (${userId}, 'personal') RETURNING id
        `);
        return { vaultId: inserted.rows[0].id, credentialIds: [] };
      }
      const creds = await tx.execute<{ id: string; version: number; updated_at: Date }>(sql`
        SELECT id, version, updated_at FROM credentials WHERE vault_id = ${v.rows[0].id} ORDER BY updated_at DESC
      `);
      return { vaultId: v.rows[0].id, credentialIds: creds.rows.map(r => ({ id: r.id, version: r.version, updatedAt: r.updated_at.toISOString() })) };
    });
  }
}
```

**`vault.controller.ts`**:
```ts
@Controller("vault")
@UseGuards(JwtAuthGuard)
export class VaultController {
  constructor(private readonly svc: VaultService) {}
  @Get("personal")
  @Throttle({ "vault-list-user": { limit: 120, ttl: 60_000 } })
  async getPersonal(@CurrentUser() user: AuthUser) {
    return this.svc.getPersonal(user.id);
  }
}
```
