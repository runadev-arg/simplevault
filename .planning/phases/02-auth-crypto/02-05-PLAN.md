Schema invariants that the auditor will check:
- Server stores `argon2_secret_key_hash` (the verifier from REQ-CRYPTO-003), NOT the raw secret_key.
- Server stores `wrapped_master_dek` and `wrapped_master_dek_recovery` (ciphertexts), NOT plaintext DEKs.
- Server stores `recovery_hmac` (HMAC-SHA256 of sha256-of-phrase), NOT the phrase or its plain hash.
- Refresh tokens stored as BLAKE2b-256 hash, with `family_id` for lineage tracking.

**`users.ts`** — extend the Phase 01 stub:
```ts
import { pgTable, text, timestamp, uuid, jsonb, customType, uniqueIndex } from "drizzle-orm/pg-core";

const bytea = customType<{ data: Uint8Array; driverData: Buffer }>({
  dataType: () => "bytea",
  toDriver: (v) => Buffer.from(v),
  fromDriver: (v) => new Uint8Array(v as Buffer),
});

export const users = pgTable("users", {
  id: uuid("id").primaryKey().defaultRandom(),
  email: text("email").notNull(),
  argon2SecretKeyHash: bytea("argon2_secret_key_hash").notNull(),
  argon2Params: jsonb("argon2_params").$type<{ memoryKiB: number; iterations: number; parallelism: 1 }>().notNull(),
  serverArgonSalt: bytea("server_argon_salt").notNull(),    // for the secret_key verifier
  userArgonSalt: bytea("user_argon_salt").notNull(),        // bound into master_KEK derivation per CRYPTO-STACK §3
  wrappedMasterDek: bytea("wrapped_master_dek").notNull(),
  wrappedMasterDekRecovery: bytea("wrapped_master_dek_recovery").notNull(),
  recoveryHmac: bytea("recovery_hmac").notNull(),
  userPubKey: bytea("user_pub_key").notNull(),               // X25519 32B
  wrappedUserSigningSk: bytea("wrapped_user_signing_sk").notNull(), // Ed25519 wrapped
  wrappedUserKxSk: bytea("wrapped_user_kx_sk").notNull(),    // X25519 wrapped
  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
}, (t) => ({
  emailLowerIdx: uniqueIndex("users_email_lower_idx").on(/* lowered email */ /* expression index */ ),
  recoveryHmacIdx: uniqueIndex("users_recovery_hmac_idx").on(t.recoveryHmac),
}));

export type User = typeof users.$inferSelect;
export type NewUser = typeof users.$inferInsert;
```
(Note: Drizzle expression-index syntax may need raw SQL; if so, generate a SQL hand-written portion of the migration. The functional uniqueness on lowercased email is what defends case-variant duplicate accounts.)

**`user_sessions.ts`**:
```ts
export const userSessions = pgTable("user_sessions", {
  id: uuid("id").primaryKey().defaultRandom(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  familyId: uuid("family_id").notNull(),                // session lineage; new on login
  refreshTokenHash: bytea("refresh_token_hash").notNull(), // BLAKE2b-256
  prevTokenId: uuid("prev_token_id"),                   // self-ref FK; null on login, set on rotation
  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  usedAt: timestamp("used_at", { withTimezone: true }),  // set on rotation
  revokedAt: timestamp("revoked_at", { withTimezone: true }), // set on logout/reuse-detect
  userAgentFamily: text("user_agent_family"),
  ipHash: bytea("ip_hash"),
  expiresAt: timestamp("expires_at", { withTimezone: true }).notNull(),
}, (t) => ({
  refreshHashIdx: uniqueIndex("user_sessions_refresh_hash_idx").on(t.refreshTokenHash),
  familyIdx: index("user_sessions_family_idx").on(t.familyId),
}));
```
