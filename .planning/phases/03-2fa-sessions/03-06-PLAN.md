async list(userId: string): Promise<TwoFaMethod[]> {
  const wa = await db.select({...}).from(webauthnCredentials).where(eq(webauthnCredentials.userId, userId));
  const totp = await db.select({...}).from(totpCredentials).where(eq(totpCredentials.userId, userId));
  return [
    ...wa.map(c => ({ id: c.id, kind: "webauthn" as const, name: c.name, createdAt: c.createdAt.toISOString(), lastUsedAt: c.lastUsedAt?.toISOString() ?? null })),
    ...totp.map(c => ({ id: c.id, kind: "totp" as const, name: c.name, createdAt: c.createdAt.toISOString(), lastUsedAt: c.lastUsedAt?.toISOString() ?? null })),
  ].sort((a, b) => (a.kind === b.kind ? a.createdAt.localeCompare(b.createdAt) : a.kind === "webauthn" ? -1 : 1));
}

async remove(userId: string, methodId: string): Promise<{kind: "webauthn"|"totp"} | null> {
  const wa = await db.delete(webauthnCredentials).where(and(eq(webauthnCredentials.id, methodId), eq(webauthnCredentials.userId, userId))).returning({ id: webauthnCredentials.id });
  if (wa.length > 0) return { kind: "webauthn" };
  const totp = await db.delete(totpCredentials).where(and(eq(totpCredentials.id, methodId), eq(totpCredentials.userId, userId))).returning({ id: totpCredentials.id });
  if (totp.length > 0) return { kind: "totp" };
  return null;
}

async countActive(userId: string): Promise<number> {
  const wa = await db.select({ c: count() }).from(webauthnCredentials).where(eq(webauthnCredentials.userId, userId));
  const totp = await db.select({ c: count() }).from(totpCredentials).where(eq(totpCredentials.userId, userId));
  return (wa[0]?.c ?? 0) + (totp[0]?.c ?? 0);
}
```

/** Phase-07 hand-off seam. Phase 03 always returns false (no shared vaults yet).
 *  Phase 07 will replace with: SELECT EXISTS(SELECT 1 FROM vault_members WHERE user_id = $1) */
export async function userHasSharedVaultDependency(_db: DbService, _userId: string): Promise<boolean> {
  return false;
}

@Injectable()
export class Require2FAGuard implements CanActivate {
  constructor(private readonly methods: MethodsService) {}
  async canActivate(ctx: ExecutionContext): Promise<boolean> {
    const req = ctx.switchToHttp().getRequest<{ user?: { id?: string } }>();
    const userId = req.user?.id;
    if (!userId) throw new HttpException({ error: { code: ErrorCodes.AUTH_INVALID_CREDENTIALS, message: "Unauthorized" } }, 401);
    const c = await this.methods.countActive(userId);
    if (c < 1) {
      throw new HttpException({ error: { code: ErrorCodes.AUTH_2FA_REQUIRED, message: "2FA enrollment required" } }, 403);
    }
    return true;
  }
}
```
