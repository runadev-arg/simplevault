**`session-epoch.cache.ts`**:
```ts
@Injectable()
export class SessionEpochCache implements OnModuleInit {
  private redis!: Redis;
  private inflight = new Map<string, Promise<number>>();
  private cacheTtlSec = 60;

  constructor(private readonly db: DbService, private readonly config: ConfigService) {}

  onModuleInit() {
    const url = this.config.get<string>("REDIS_URL") ?? process.env.REDIS_URL ?? "redis://localhost:6379";
    this.redis = new Redis(url, { lazyConnect: true, maxRetriesPerRequest: 1, enableOfflineQueue: false });
    this.redis.on("error", () => {});
    const ttlEnv = process.env.SESSION_EPOCH_CACHE_TTL;
    const n = ttlEnv ? Number.parseInt(ttlEnv, 10) : NaN;
    this.cacheTtlSec = Number.isFinite(n) && n > 0 ? n : 60;
  }

  async get(userId: string): Promise<number> {
    const key = `session-epoch:${userId}`;
    try {
      const cached = await this.redis.get(key);
      if (cached !== null) return Number.parseInt(cached, 10);
    } catch { /* fail open — DB read */ }
    let p = this.inflight.get(userId);
    if (!p) {
      p = (async () => {
        const rows = await this.db.db.select({ e: schema.users.sessionEpoch }).from(schema.users).where(eq(schema.users.id, userId)).limit(1);
        const epoch = rows[0]?.e ?? 0;
        try { await this.redis.set(key, epoch.toString(), "EX", this.cacheTtlSec); } catch { /* swallow */ }
        return epoch;
      })().finally(() => this.inflight.delete(userId));
      this.inflight.set(userId, p);
    }
    return p;
  }

  async bust(userId: string): Promise<void> {
    const key = `session-epoch:${userId}`;
    try { await this.redis.del(key); } catch { /* swallow — DB read on next request */ }
  }
}
```
