**`enroll-totp-flow.tsx`**:
```tsx
"use client";
import { computeTotpStep, verifyTotpCandidate, buildOtpauthUrl } from "@simplevault/crypto";
import { aeadEncrypt } from "@simplevault/crypto/browser";  // existing helper
import sodium from "libsodium-wrappers-sumo";
import QRCode from "qrcode";

export function EnrollTotpFlow() {
  const [step, setStep] = useState<"init" | "show-qr" | "verify" | "done">("init");
  const [secret, setSecret] = useState<Uint8Array | null>(null);
  const [issuanceNonce, setIssuanceNonce] = useState<string | null>(null);
  const [qrDataUrl, setQrDataUrl] = useState("");
  const [code, setCode] = useState("");
  const [name, setName] = useState("Authenticator");
  // ...
  async function start() {
    const { issuanceNonce } = await beginTotpRegister();
    await sodium.ready;
    const sec = sodium.randombytes_buf(20);  // 20 bytes per RFC 6238 norm; 160-bit key
    const url = buildOtpauthUrl({ issuer: "SimpleVault", account: userEmail, secret: sec });
    const qr = await QRCode.toDataURL(url);
    setSecret(sec); setIssuanceNonce(issuanceNonce); setQrDataUrl(qr);
    setStep("show-qr");
  }
  async function verify() {
    const currentStep = Math.floor(Date.now() / 1000 / 30);
    const v = verifyTotpCandidate(secret!, code, currentStep, 1);
    if (!v.ok) { /* surface "Wrong code, try again" */ return; }
    // Wrap secret under master_DEK
    const aad = aadFor("sv:user-totp:v1|", userEmail);  // helper from existing aad-labels module
    const wrapped = await aeadEncrypt(masterDEK, sec, aad);
    await finishTotpRegister({ issuanceNonce, wrappedSecret: bytesToBase64(wrapped), encryptedSecretAad: bytesToBase64(aad), name, candidateStep: v.step });
    // Wipe locals
    sec.fill(0); setSecret(null); setIssuanceNonce(null); setQrDataUrl("");
    setStep("done");
  }
}
```

(`bytesToBase64`, `aadFor` are existing helpers in `apps/web/src/lib/crypto/...`; reuse.)

**`step-up-flow.ts`**:
```ts
export async function completeWithPasskey(stepUpToken: string): Promise<SessionResult> {
  const opts = await beginWebauthnAuth(stepUpToken);
  const ass = await startAuthentication({ optionsJSON: opts });
  return await finishWebauthnAuth(stepUpToken, ass);  // returns { accessToken, expiresAt, wrappedMaterial, refreshCookie set by server }
}
export async function completeWithTotp(stepUpToken: string, credentialId: string, code: string, secret: Uint8Array): Promise<SessionResult> {
  const currentStep = Math.floor(Date.now() / 1000 / 30);
  const v = verifyTotpCandidate(secret, code, currentStep, 1);
  if (!v.ok) throw new Error("Invalid code");
  return await verifyTotpApi(stepUpToken, { credentialId, candidateStep: v.step });
}
```

Note: completing with TOTP requires the user's TOTP secret which lives WRAPPED in `totp_credentials.wrapped_secret`. The login flow must list the user's TOTP credentials (added field on the step-up response? OR fetch via a new auth-skipping endpoint protected by the step-up token: `GET /2fa/totp/credentials` that returns `[{id, name, wrappedSecret, encryptedSecretAad}]` for the step-up's userId). Picked the latter — cleaner; one new endpoint added in this commit (`apps/api/src/twofa/totp/totp.controller.ts:GET /2fa/totp/credentials` guarded by Require2FAStepUpGuard, returns the wrap blobs so the client can decrypt-verify-locally). Document this added endpoint in the SUMMARY.

But wait — the master_DEK is needed to decrypt the wrapped secret. master_DEK is derived from master_password + secret_key. So the user has already provided those at step-up issuance (the 1FA login). The keyStore has master_DEK from that flow. Good — completeWithTotp can read from keyStore.
