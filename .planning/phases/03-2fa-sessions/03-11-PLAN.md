```tsx
"use client";
import { revokeAllSessions } from "@/lib/api/sessions-client";
import { keyStore, accessTokenStore } from "@/lib/auth/key-store";

export function RevokeAllButton() {
  const [busy, setBusy] = useState(false);
  const [confirming, setConfirming] = useState(false);
  if (!confirming) {
    return <button onClick={() => setConfirming(true)} className="text-rose-700 ...">Sign out everywhere except this device</button>;
  }
  return (
    <div>
      <p>This will sign you out of all your devices, including this one. You'll need to log in again here.</p>
      <button disabled={busy} onClick={async () => {
        setBusy(true);
        try {
          await revokeAllSessions();  // 200 + sets clearing cookie
          keyStore.wipe();            // zero out master_DEK + private keys
          accessTokenStore.clear();
          window.location.assign("/login");  // hard redirect — full page wipe
        } catch {
          // even on failure, do defence-in-depth wipe
          keyStore.wipe(); accessTokenStore.clear();
          window.location.assign("/login");
        }
      }}>Confirm — sign out everywhere</button>
      <button onClick={() => setConfirming(false)}>Cancel</button>
    </div>
  );
}
```

Defence-in-depth wipe-on-API-failure mirrors the existing logout pattern in `auth-context.tsx` (Phase 02-11).
