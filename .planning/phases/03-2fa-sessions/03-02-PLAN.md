Step-up flow: when a user with 2FA enabled hits `/auth/login`, they get a step-up token instead of a full session. They redeem it via `/2fa/webauthn/begin-auth` + `/finish-auth` (or via `/2fa/totp/verify` from Plan 03). Only `/finish-auth` and `/2fa/totp/verify` mint full sessions.

export interface StepUpClaims { sub: string; purpose: "2fa-stepup"; epoch: number; }

@Injectable()
export class StepUpJwtService implements OnModuleInit {
  private secret!: Uint8Array;
  private ttlSec!: number;
  constructor(private readonly jwt: JwtService, private readonly config: ConfigService) {}
  onModuleInit(): void {
    this.secret = this.jwt.exposeSecret(); // add a friend method on JwtService
    const raw = this.config.get<string>("STEP_UP_TOKEN_TTL") ?? process.env.STEP_UP_TOKEN_TTL;
    const n = raw ? Number.parseInt(raw, 10) : NaN;
    this.ttlSec = Number.isFinite(n) && n > 0 ? n : 120;
  }
  async sign(userId: string, epoch: number): Promise<string> { /* jose SignJWT, alg=HS256, kid=primary, payload {sub:userId, purpose:'2fa-stepup', epoch}, exp = iat + ttlSec */ }
  async verify(token: string): Promise<StepUpClaims> { /* jose jwtVerify, assert payload.purpose === '2fa-stepup', else throw */ }
}
```

async beginRegister(userId: string, userEmail: string): Promise<PublicKeyCredentialCreationOptionsJSON> {
  const opts = await generateRegistrationOptions({
    rpName: WEBAUTHN_RP_NAME,
    rpID: WEBAUTHN_RP_ID,
    userID: Buffer.from(userId, "utf8"),
    userName: userEmail,
    attestationType: "none",
    authenticatorSelection: { userVerification: "required", residentKey: "preferred" },
    supportedAlgorithmIDs: [-7, -257], // ES256 + RS256
    excludeCredentials: existingCreds.map(c => ({ id: c.credentialId, type: "public-key", transports: c.transports })),
  });
  // Write challenge row, ON CONFLICT (user_id, kind) DO UPDATE SET challenge=excluded.challenge, expires_at=excluded.expires_at
  await db.insert(webauthnChallenges).values({ userId, kind: "register", challenge: Buffer.from(opts.challenge, "base64url"), expiresAt: new Date(Date.now() + 120_000) }).onConflictDoUpdate({ target: [webauthnChallenges.userId, webauthnChallenges.kind], set: { challenge: ..., expiresAt: ..., createdAt: new Date() } });
  return opts;
}

async finishRegister(userId: string, body: RegistrationResponseJSON, name: string): Promise<{id:string, name:string}> {
  // Atomic consume:
  const consumed = await db.execute<{ challenge: Buffer; expires_at: Date }>(sql`DELETE FROM webauthn_challenges WHERE user_id = ${userId} AND kind = 'register' RETURNING challenge, expires_at`);
  const row = consumed.rows[0];
  if (!row) throwWebauthnInvalid();
  if (row.expires_at <= new Date()) throwWebauthnInvalid();
  const verification = await verifyRegistrationResponse({
    response: body,
    expectedChallenge: row.challenge.toString("base64url"),
    expectedOrigin: WEBAUTHN_ORIGIN,
    expectedRPID: WEBAUTHN_RP_ID,
    requireUserVerification: true,
  });
  if (!verification.verified || !verification.registrationInfo) throwWebauthnVerificationFailed();
  const ri = verification.registrationInfo;
  const inserted = await db.insert(webauthnCredentials).values({
    userId,
    credentialId: Buffer.from(ri.credential.id),
    publicKey: Buffer.from(ri.credential.publicKey),
    counter: ri.credential.counter,
    transports: body.response.transports ?? [],
    aaguid: ri.aaguid ? Buffer.from(ri.aaguid) : null,
    name,
  }).returning({ id: webauthnCredentials.id, name: webauthnCredentials.name });
  AuditEventService.emit(this.logger, { action: "auth.2fa.webauthn.register.ok", actorUserId: userId, targetId: inserted[0].id, outcome: "ok" });
  return inserted[0];
}
```

**`WebauthnAuthService`**:
```ts
async beginAuth(userId: string): Promise<PublicKeyCredentialRequestOptionsJSON> {
  const creds = await db.select(...).from(webauthnCredentials).where(eq(webauthnCredentials.userId, userId));
  if (creds.length === 0) throw new HttpException({ error: { code: "AUTH_2FA_NO_METHOD", message: "No 2FA method enrolled" } }, 400);
  const opts = await generateAuthenticationOptions({
    rpID: WEBAUTHN_RP_ID,
    userVerification: "required",
    allowCredentials: creds.map(c => ({ id: c.credentialId, type: "public-key", transports: c.transports })),
  });
  await db.insert(webauthnChallenges)...onConflictDoUpdate(...) // kind='auth'
  return opts;
}

async finishAuth(userId: string, body: AuthenticationResponseJSON, ip: string, ua: string|undefined): Promise<{accessToken: string; refreshCookie: string; wrappedMaterial: ...}> {
  const consumed = await db.execute<{...}>(sql`DELETE FROM webauthn_challenges WHERE user_id = ${userId} AND kind = 'auth' RETURNING challenge, expires_at`);
  const ch = consumed.rows[0]; if (!ch || ch.expires_at <= new Date()) throwInvalid();
  const cred = await db.select(...).from(webauthnCredentials).where(and(eq(webauthnCredentials.userId, userId), eq(webauthnCredentials.credentialId, base64urlToBuffer(body.id))));
  if (!cred[0]) throwInvalid();
  const verification = await verifyAuthenticationResponse({
    response: body,
    expectedChallenge: ch.challenge.toString("base64url"),
    expectedOrigin: WEBAUTHN_ORIGIN,
    expectedRPID: WEBAUTHN_RP_ID,
    credential: { id: cred[0].credentialId, publicKey: cred[0].publicKey, counter: cred[0].counter, transports: cred[0].transports },
    requireUserVerification: true,
  });
  if (!verification.verified) { audit.fail("auth.2fa.webauthn.auth.fail", "verification_failed"); throwVerifFailed(); }
  // Counter regression: only check when stored.counter > 0 (some authenticators never increment, returning 0)
  const newCounter = verification.authenticationInfo.newCounter;
  if (cred[0].counter > 0 && newCounter <= cred[0].counter) {
    audit.fail("auth.2fa.webauthn.auth.fail", "counter_regression");
    throwVerifFailed();
  }
  await db.update(webauthnCredentials).set({ counter: newCounter, lastUsedAt: new Date() }).where(eq(webauthnCredentials.id, cred[0].id));
  // Mint full session — delegate to SessionService.createOnLogin (existing Phase-02 primitive)
  const issued = await this.sessions.createOnLogin(userId, ip, ua);
  const accessToken = await this.jwt.signAccessToken({ sub: userId, sid: issued.sessionId, fam: issued.familyId, epoch: /* Plan 04 will populate */ 0 });
  audit.ok("auth.2fa.webauthn.auth.ok", userId);
  return { accessToken, rawRefreshToken: issued.rawToken, expiresAt: issued.expiresAt };
}
```
