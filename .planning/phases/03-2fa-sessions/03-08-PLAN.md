**`auth-client.ts`** — extend the `login(...)` function. Currently returns Phase-02's success shape. Change to:
```ts
export type LoginResult =
  | { kind: "session"; accessToken: string; expiresAt: number; wrappedMaterial: ... }
  | { kind: "step-up"; stepUpToken: string; twoFa: { webauthnAvailable: boolean; totpAvailable: boolean } };

export async function login(email: string, secretKey: string, masterPassword: string): Promise<LoginResult> {
  const params = await fetch("/api/auth/params").then(r => r.json());
  const candidateHash = await argon2idVerifier(secretKey, params.argon2Params);
  const res = await fetch("/api/auth/login", { method: "POST", headers: {...}, body: JSON.stringify({ email, argon2SecretKeyHash: candidateHash }) });
  if (!res.ok) throw new AuthClientError(...);
  const body = await res.json();
  if (body.kind === "2fa-required") return { kind: "step-up", stepUpToken: body.stepUpToken, twoFa: body.twoFa };
  // existing path
  return { kind: "session", accessToken: body.accessToken, expiresAt: ..., wrappedMaterial: ... };
}
```
