**Implementation**:
```ts
import sodium from "libsodium-wrappers-sumo";

await sodium.ready;

export function computeTotpStep(secret: Uint8Array, step: number, digits = 6): string {
  // 8-byte big-endian step counter
  const buf = new Uint8Array(8);
  const view = new DataView(buf.buffer);
  view.setBigUint64(0, BigInt(step), false);
  // libsodium's HMAC: crypto_auth_hmacsha1 (32-byte key max — RFC accepts longer; truncate per RFC 4226)
  // Actually libsodium-wrappers-sumo exposes crypto_auth_hmacsha1_BYTES = 20.
  const mac = sodium.crypto_auth_hmacsha1(buf, secret); // returns 20-byte HMAC
  // Dynamic truncation per RFC 4226 §5.3
  const offset = mac[mac.length - 1] & 0x0f;
  const code =
    ((mac[offset] & 0x7f) << 24) |
    ((mac[offset + 1] & 0xff) << 16) |
    ((mac[offset + 2] & 0xff) << 8) |
    (mac[offset + 3] & 0xff);
  const mod = 10 ** digits;
  return (code % mod).toString().padStart(digits, "0");
}

export function verifyTotpCandidate(secret: Uint8Array, code: string, currentStep: number, drift = 1): { ok: boolean; step?: number } {
  if (!/^\d{6}$/.test(code)) return { ok: false };
  for (let d = -drift; d <= drift; d++) {
    const candidate = computeTotpStep(secret, currentStep + d);
    if (sodium.memcmp(new TextEncoder().encode(candidate), new TextEncoder().encode(code))) {
      return { ok: true, step: currentStep + d };
    }
  }
  return { ok: false };
}

export function buildOtpauthUrl(opts: { issuer: string; account: string; secret: Uint8Array; period?: number; digits?: number }): string {
  const period = opts.period ?? 30;
  const digits = opts.digits ?? 6;
  const b32 = base32encode(opts.secret); // RFC 4648 base32 (no padding)
  const label = encodeURIComponent(`${opts.issuer}:${opts.account}`);
  const params = new URLSearchParams({
    secret: b32,
    issuer: opts.issuer,
    algorithm: "SHA1",
    digits: digits.toString(),
    period: period.toString(),
  });
  return `otpauth://totp/${label}?${params.toString()}`;
}

// base32 helper — RFC 4648; alphabet "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"
```

**`TotpService.verify(userId, dto)`** where dto = `{credentialId, candidateStep}`:

```sql
UPDATE totp_credentials
   SET last_used_step = $cs, last_used_at = now()
 WHERE id = $cid AND user_id = $uid AND last_used_step < $cs
RETURNING id;
```
- Zero rows → 401 `AUTH_2FA_TOTP_REPLAY` + audit fail.
