**docker-compose.yml**:
```yaml
name: simplevault

x-restart: &restart
  restart: unless-stopped

services:
  postgres:
    image: postgres:18.3-alpine
    <<: *restart
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-simplevault}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-changeme_dev_only}
      POSTGRES_DB: ${POSTGRES_DB:-simplevault}
      PGDATA: /var/lib/postgresql/data/pgdata
    volumes:
      - pg_data:/var/lib/postgresql/data
    networks: [backend]
    # NO host port mapping — backend network only.
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 15s
    mem_limit: 512m
    pids_limit: 200
    security_opt:
      - no-new-privileges:true

  redis:
    image: redis:7.4-alpine
    <<: *restart
    command: ["redis-server", "--appendonly", "yes", "--save", "60", "1000"]
    volumes:
      - redis_data:/data
    networks: [backend]
    healthcheck:
      test: ["CMD-SHELL", "redis-cli ping | grep -q PONG"]
      interval: 10s
      timeout: 3s
      retries: 5
    mem_limit: 256m
    pids_limit: 100
    security_opt:
      - no-new-privileges:true

  api:
    build:
      context: .
      dockerfile: apps/api/Dockerfile
    <<: *restart
    environment:
      NODE_ENV: development
      PORT: 3001
      DATABASE_URL: postgres://${POSTGRES_USER:-simplevault}:${POSTGRES_PASSWORD:-changeme_dev_only}@postgres:5432/${POSTGRES_DB:-simplevault}
      REDIS_URL: redis://redis:6379
      LOG_LEVEL: debug
      CORS_ORIGINS: http://localhost:3000
      JWT_SECRET: ${JWT_SECRET:-dev_only_replace_in_prod_32bytes_minimum_______}
      SERVER_CHAIN_SECRET: ${SERVER_CHAIN_SECRET:-dev_only_audit_chain_hmac_secret_32bytes_min______}
    depends_on:
      postgres: { condition: service_healthy }
      redis: { condition: service_healthy }
    ports:
      - "127.0.0.1:3001:3001"  # Localhost only, for /health checks during dev
    networks: [backend, frontend]
    mem_limit: 512m
    pids_limit: 200
    security_opt:
      - no-new-privileges:true
    read_only: false  # NestJS writes to /tmp; keep as-is for now. Phase 12 may tighten with tmpfs.
    cap_drop: [ALL]

  web:
    build:
      context: .
      dockerfile: apps/web/Dockerfile
    <<: *restart
    environment:
      NODE_ENV: development
      PORT: 3000
      HOSTNAME: 0.0.0.0
      NEXT_PUBLIC_API_URL: http://localhost:3001
    depends_on:
      api: { condition: service_started }
    ports:
      - "127.0.0.1:3000:3000"
    networks: [frontend]
    mem_limit: 512m
    pids_limit: 200
    security_opt:
      - no-new-privileges:true
    cap_drop: [ALL]

volumes:
  pg_data:
  redis_data:

networks:
  backend:
    driver: bridge
    internal: true   # CRITICAL: postgres + redis cannot reach internet
  frontend:
    driver: bridge
```

Note on `cap_drop: [ALL]`: Node.js doesn't need any Linux capabilities for its server work. Helmet etc. are pure userspace.

Note on `internal: true` backend network + api needs both networks: api is the bridge — connected to backend (talks to pg/redis) AND frontend (talks to web/host). pg and redis are ONLY on backend.

**.env.example**:
```bash
# ----------------------------------------------------------------------------
# SimpleVault — environment variables
# Copy to `.env` for local dev (or use Dokploy's encrypted env-var UI in prod).
# NEVER commit a real `.env`.
# ----------------------------------------------------------------------------

# ----- Postgres -----
# In local dev (docker compose), defaults below are OK.
# In prod (Dokploy managed Postgres 18.3), Dokploy provides the connection
# string — set DATABASE_URL directly to that and ignore POSTGRES_*.
POSTGRES_USER=simplevault
POSTGRES_PASSWORD=changeme_dev_only
POSTGRES_DB=simplevault

# ----- Database connection (used by api) -----
# Local docker-compose default (overridden in prod):
DATABASE_URL=postgres://simplevault:changeme_dev_only@postgres:5432/simplevault

# ----- Redis (used by api for rate limiting + sessions) -----
REDIS_URL=redis://redis:6379

# ----- JWT signing (REQUIRED in prod) -----
# Min 32 bytes (256 bits). Generate: `openssl rand -base64 48`
JWT_SECRET=dev_only_replace_in_prod_32bytes_minimum_______

# ----- Audit log HMAC chain secret (REQUIRED in prod) -----
# Min 32 bytes. Used to compute the per-vault hash chain HMAC.
# DO NOT rotate without coordinating with the audit chain re-bootstrap procedure
# (to be defined in Phase 10).
SERVER_CHAIN_SECRET=dev_only_audit_chain_hmac_secret_32bytes_min______

# ----- CORS allowlist (CSV, no spaces) -----
# Local: just the web dev server. Prod: https://pass.runadev.com (no trailing slash).
CORS_ORIGINS=http://localhost:3000

# ----- Web -> API URL (public, used by browser) -----
# Local: http://localhost:3001. Prod: https://api.pass.runadev.com OR same-origin via reverse proxy.
NEXT_PUBLIC_API_URL=http://localhost:3001

# ----- Logging -----
LOG_LEVEL=debug   # use `info` in prod
NODE_ENV=development

# ----- Node runtime port (rarely overridden) -----
PORT=3001         # api
# WEB_PORT defaults to 3000 (set in apps/web Dockerfile)

# ----------------------------------------------------------------------------
# Phase 02+ will add:
#   ARGON2_MEM_KIB, ARGON2_ITERATIONS, ARGON2_PARALLELISM
#   COOKIE_DOMAIN, COOKIE_SAMESITE
#   SMTP_* (Phase 02 for invite/login alerts; Phase 07 for vault invites)
#   TURNSTILE_* or HCAPTCHA_* (Phase 02 if we add captcha to signup)
# ----------------------------------------------------------------------------
```
