**apps/api/Dockerfile**:
```dockerfile
# syntax=docker/dockerfile:1.7
ARG NODE_VERSION=22-alpine

# ---------- deps ----------
FROM node:${NODE_VERSION} AS deps
RUN apk add --no-cache libc6-compat
ENV PNPM_HOME=/pnpm
ENV PATH=$PNPM_HOME:$PATH
RUN corepack enable && corepack prepare pnpm@9.15.0 --activate

WORKDIR /repo
COPY pnpm-workspace.yaml pnpm-lock.yaml package.json .npmrc ./
COPY turbo.json tsconfig.base.json ./
COPY packages/tsconfig/package.json packages/tsconfig/
COPY packages/eslint-config/package.json packages/eslint-config/
COPY packages/shared/package.json packages/shared/
COPY packages/crypto/package.json packages/crypto/
COPY packages/db/package.json packages/db/
COPY apps/api/package.json apps/api/

# Use pnpm fetch for cache-friendly install
RUN --mount=type=cache,id=pnpm-store,target=/pnpm/store \
    pnpm install --frozen-lockfile --prod=false

# ---------- build ----------
FROM deps AS build
WORKDIR /repo
COPY packages packages
COPY apps/api apps/api
RUN pnpm --filter @simplevault/shared build \
 && pnpm --filter @simplevault/crypto build \
 && pnpm --filter @simplevault/db build \
 && pnpm --filter @simplevault/api build

# Prune to prod deps
RUN --mount=type=cache,id=pnpm-store,target=/pnpm/store \
    pnpm --filter @simplevault/api deploy --prod /out

# ---------- runner ----------
FROM node:${NODE_VERSION} AS runner
RUN apk add --no-cache tini curl \
 && addgroup -S app && adduser -S -G app -h /app app

WORKDIR /app
COPY --from=build --chown=app:app /out/node_modules ./node_modules
COPY --from=build --chown=app:app /out/dist ./dist
COPY --from=build --chown=app:app /repo/packages/db/drizzle ./packages/db/drizzle
COPY --from=build --chown=app:app /repo/packages/db/drizzle.config.ts ./packages/db/drizzle.config.ts
COPY --from=build --chown=app:app /repo/packages/db/dist ./packages/db/dist
COPY --chown=app:app apps/api/scripts/migrate-then-start.sh ./migrate-then-start.sh
RUN chmod +x ./migrate-then-start.sh

USER app
ENV NODE_ENV=production
ENV PORT=3001

EXPOSE 3001
HEALTHCHECK --interval=20s --timeout=5s --start-period=15s --retries=3 \
  CMD curl -fsS http://localhost:3001/health || exit 1

ENTRYPOINT ["/sbin/tini", "--", "./migrate-then-start.sh"]
```

**apps/api/scripts/migrate-then-start.sh**:
```sh
#!/bin/sh
set -eu

echo "[startup] Running drizzle migrate..."
cd /app/packages/db
# drizzle-kit lives in deps because we built with --prod=false initially.
# After --prod prune, drizzle-kit is gone. Solution: copy a tiny migrate runner.
node ./dist/migrate.js
cd /app

echo "[startup] Migrations applied. Starting api..."
exec node dist/main.js
```

For this to work, **add a tiny custom migrate runner** at `packages/db/src/migrate.ts`:
```ts
import { migrate } from "drizzle-orm/node-postgres/migrator";
import { createDbClient } from "./client.js";

const url = process.env.DATABASE_URL;
if (!url) { console.error("DATABASE_URL is required"); process.exit(1); }

const { db, pool } = createDbClient({ connectionString: url });
try {
  await migrate(db, { migrationsFolder: "./drizzle" });
  console.log("Migrations applied");
} catch (err) {
  console.error("Migration failed:", err);
  process.exit(1);
} finally {
  await pool.end();
}
```

Update `packages/db/src/index.ts` exports? No — `migrate.ts` is invoked directly via `node ./dist/migrate.js`, not imported. Just ensure tsconfig `include` covers it.

`packages/db/tsconfig.json` `include`: should already cover `src/**/*`, fine.

Note on `apps/web/server.js`: Next.js standalone outputs the server entry there. The Dockerfile assumes that path is correct given monorepo structure (Next 15 in monorepos: `outputFileTracingRoot: path.join(__dirname, '../../')` may be needed in `next.config.mjs`).

**Add to `apps/web/next.config.mjs`**:
```js
import path from "node:path";
import { fileURLToPath } from "node:url";
const __dirname = path.dirname(fileURLToPath(import.meta.url));

/** @type {import('next').NextConfig} */
const nextConfig = {
  output: "standalone",
  outputFileTracingRoot: path.join(__dirname, "../../"),
  reactStrictMode: true,
  poweredByHeader: false,
  experimental: { typedRoutes: true },
};
export default nextConfig;
```
